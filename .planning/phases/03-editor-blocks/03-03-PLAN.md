**Shared meta hook pattern:**
```javascript
const postType = useSelect(
    ( select ) => select( 'core/editor' ).getCurrentPostType(),
    []
);
const [ meta, setMeta ] = useEntityProp( 'postType', postType, 'meta' );
if ( ! meta ) return null;
```

CRITICAL: Always spread: `setMeta( { ...meta, 'key': value } )` — never just `{ key: value }`.

**Panel 2 — Talk Reference** (name: "speekr-conf-talk", title: "Talk Reference"):
Search-as-you-type for selecting a single Talk. Pattern:

```javascript
// Inside the panel component
const [ talkSearch, setTalkSearch ] = useState( '' );
const [ talkResults, setTalkResults ] = useState( [] );
const talkDebounceRef = useRef( null );
const selectedTalkId = meta._speekr_conf_talk_ref ?? 0;
const [ selectedTalkTitle, setSelectedTalkTitle ] = useState( '' );

// Fetch title of currently selected talk on mount
useEffect( () => {
    if ( ! selectedTalkId ) return;
    apiFetch( { path: `/wp/v2/talks/${ selectedTalkId }?_fields=id,title` } )
        .then( ( post ) => setSelectedTalkTitle( post.title.rendered ) )
        .catch( () => {} );
}, [ selectedTalkId ] );

// Debounced search
useEffect( () => {
    if ( talkSearch.length < 2 ) { setTalkResults( [] ); return; }
    clearTimeout( talkDebounceRef.current );
    talkDebounceRef.current = setTimeout( () => {
        apiFetch( {
            path: addQueryArgs( '/wp/v2/talks', {
                search: talkSearch,
                per_page: 10,
                status: 'publish',
                _fields: 'id,title',
            } ),
        } ).then( setTalkResults ).catch( () => {} );
    }, 300 );
}, [ talkSearch ] );
```

Render:
- If selectedTalkId > 0: show selected title + "Clear" Button (variant: "link", isDestructive) that sets _speekr_conf_talk_ref to 0 via setMeta spread
- TextControl for search (label: "Search Talks", value: talkSearch, onChange: setTalkSearch)
- Map talkResults → Button (variant: "tertiary") per result; clicking selects that talk: setMeta spread with _speekr_conf_talk_ref = post.id, clear search

**Panel 3 — Speakers** (name: "speekr-conf-speakers", title: "Speakers"):
Add-one-at-a-time speaker search. _speekr_conf_speakers is an array of integers (Speaker Profile post IDs).

```javascript
const [ speakerSearch, setSpeakerSearch ] = useState( '' );
const [ speakerResults, setSpeakerResults ] = useState( [] );
const [ speakerNames, setSpeakerNames ] = useState( {} ); // id -> name cache
const speakerDebounceRef = useRef( null );
const selectedSpeakers = meta._speekr_conf_speakers ?? [];

// Load names for currently selected speakers
useEffect( () => {
    if ( ! selectedSpeakers.length ) return;
    const uncached = selectedSpeakers.filter( ( id ) => ! speakerNames[ id ] );
    if ( ! uncached.length ) return;
    apiFetch( {
        path: addQueryArgs( '/wp/v2/speekr_speaker', {
            include: uncached.join( ',' ),
            _fields: 'id,title',
        } ),
    } ).then( ( posts ) => {
        const newNames = {};
        posts.forEach( ( p ) => { newNames[ p.id ] = p.title.rendered; } );
        setSpeakerNames( ( prev ) => ( { ...prev, ...newNames } ) );
    } ).catch( () => {} );
}, [ selectedSpeakers ] );

// Debounced search
useEffect( () => {
    if ( speakerSearch.length < 2 ) { setSpeakerResults( [] ); return; }
    clearTimeout( speakerDebounceRef.current );
    speakerDebounceRef.current = setTimeout( () => {
        apiFetch( {
            path: addQueryArgs( '/wp/v2/speekr_speaker', {
                search: speakerSearch,
                per_page: 10,
                status: 'publish',
                _fields: 'id,title',
            } ),
        } ).then( setSpeakerResults ).catch( () => {} );
    }, 300 );
}, [ speakerSearch ] );

const addSpeaker = ( id ) => {
    if ( selectedSpeakers.includes( id ) ) return;
    setMeta( { ...meta, _speekr_conf_speakers: [ ...selectedSpeakers, id ] } );
    setSpeakerSearch( '' );
    setSpeakerResults( [] );
};

const removeSpeaker = ( id ) => {
    setMeta( {
        ...meta,
        _speekr_conf_speakers: selectedSpeakers.filter( ( s ) => s !== id ),
    } );
};
```

Render:
- List of selected speakers: `<ul>` with `<li>` per ID showing speakerNames[id] or `#${id}` fallback + "×" Button (variant: "link", isDestructive) to remove
- TextControl for search (label: "Add Speaker", value: speakerSearch, onChange: setSpeakerSearch)
- Map speakerResults → Button (variant: "tertiary") per result; clicking calls addSpeaker(post.id)
